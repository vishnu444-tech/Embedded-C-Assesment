You are given an 8-bit register represented as an unsigned char. Write a function to: ● Set the 3rd bit (bit index 2). ● Clear the 6th bit (bit index 5). ● Toggle the 1st bit (bit index 0). Return the modified register value. Note: Use bitwise operators only. Avoid loops or conditionals.

#include <stdio.h>

unsigned char modify_register(unsigned char reg)
{
    reg |=  (1U << 2);    
    reg &= ~(1U << 5);    
    reg ^=  (1U << 0);    
    return reg;
}

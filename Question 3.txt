Write a C program to print a pyramid of stars for a given number n. If n = 5, the output should be:


#include <stdio.h>

int main() 
{
    int n;
    printf("Enter n: ");
    scanf("%d", &n);

    for (int i = 1; i <= n; i++)
 {
        // leading spaces
        for (int j = 1; j <= n - i; j++)
            printf(" ");

        // stars: 1, 3, 5, ... (2i - 1)
        for (int k = 1; k <= 2 * i - 1; k++)
            printf("*");

        printf("\n");
    }
    return 0;
}
